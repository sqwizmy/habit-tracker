Я помогу выработать привычки.", reply_markup=menu_kb()) @dp.callback_query(F.data == "menu") async def menu(c: CallbackQuery): await c.message.edit_text("Меню", reply_markup=menu_kb()) @dp.callback_query(F.data.startswith("d:")) async def day(c: CallbackQuery): text, kb = today_view(c.from_user.id, date.fromisoformat(c.data[2:])) await c.message.edit_text(text, reply_markup=kb) @dp.callback_query(F.data.startswith("t:")) async def toggle(c: CallbackQuery): _, hid, d = c.data.split(":") hid = int(hid) if not q("SELECT 1 FROM habits WHERE id=?
